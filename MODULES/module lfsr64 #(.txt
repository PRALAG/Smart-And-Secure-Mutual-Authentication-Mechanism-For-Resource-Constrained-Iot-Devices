module lfsr64 #(
    parameter SEED = 64'hACEB00C123456789
)(
    input  clk,
    input  reset,
    input  enable,
    output reg [63:0] rnd,
    output reg        rnd_valid
);
    reg [3:0] div_cnt;

    always @(posedge clk or posedge reset) begin
        if (reset) begin
            rnd       <= SEED;
            div_cnt   <= 4'd0;
            rnd_valid <= 1'b0;
        end
        else if (enable) begin
            div_cnt <= div_cnt + 1;
            if (div_cnt == 4'b1111) begin
                rnd       <= {rnd[62:0], rnd[63] ^ rnd[62]};
                rnd_valid <= 1'b1;
            end
        end
    end
endmodule
